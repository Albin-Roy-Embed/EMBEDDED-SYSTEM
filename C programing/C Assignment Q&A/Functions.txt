5.  Write a C program that uses a function to check whether a given number is a 
Perfect Number. 

#include <stdio.h>

int isPerfect(int num) {
    int sum = 0;
    for (int i = 1; i < num; i++) {
        if (num % i == 0) {
            sum += i;
        }
    }
    return (sum == num);
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);

    if (isPerfect(n))
        printf("%d is a Perfect Number\n", n);
    else
        printf("%d is not a Perfect Number\n", n);

    return 0;
}
