21. Write a C program to remove duplicate elements from a given array. 

#include <stdio.h>

int main() {
    int arr[100], n, i, j, k;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    // Remove duplicates
    for (i = 0; i < n; i++) {
        for (j = i + 1; j < n; j++) {
            if (arr[i] == arr[j]) {
                for (k = j; k < n - 1; k++) {
                    arr[k] = arr[k + 1];
                }
                n--;   // shrink size
                j--;   // stay at same index
            }
        }
    }

    printf("Unique array:\n");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}

23. Write a C program that rotates the elements of an array by n positions in a 
specified direction.

#include <stdio.h>

int main() {
    int arr[100], n, size, i;
    char dir;

    printf("Enter size: ");
    scanf("%d", &size);

    printf("Enter %d elements:\n", size);
    for (i = 0; i < size; i++) scanf("%d", &arr[i]);

    printf("Rotate by how many positions: ");
    scanf("%d", &n);

    printf("Direction (L/R): ");
    scanf(" %c", &dir);

    n = n % size;  // normalize

    int res[100];
    for (i = 0; i < size; i++) {
        if (dir == 'L' || dir == 'l')
            res[i] = arr[(i + n) % size];
        else
            res[i] = arr[(i - n + size) % size];
    }

    printf("Rotated array:\n");
    for (i = 0; i < size; i++) printf("%d ", res[i]);

    return 0;
}


24. Write a C program that merges two sorted arrays into a single sorted array 
without using any sorting function. 

#include <stdio.h>

int main() {
    int arr1[100], arr2[100], merged[200];
    int n1, n2, i = 0, j = 0, k = 0;

    printf("Enter size of first array: ");
    scanf("%d", &n1);
    printf("Enter %d sorted elements:\n", n1);
    for (i = 0; i < n1; i++) scanf("%d", &arr1[i]);

    printf("Enter size of second array: ");
    scanf("%d", &n2);
    printf("Enter %d sorted elements:\n", n2);
    for (i = 0; i < n2; i++) scanf("%d", &arr2[i]);

    i = j = 0;

    // Merge process
    while (i < n1 && j < n2) {
        if (arr1[i] <= arr2[j])
            merged[k++] = arr1[i++];
        else
            merged[k++] = arr2[j++];
    }

    // Copy remaining elements
    while (i < n1) merged[k++] = arr1[i++];
    while (j < n2) merged[k++] = arr2[j++];

    printf("Merged sorted array:\n");
    for (i = 0; i < k; i++) printf("%d ", merged[i]);

    return 0;
}

25. Write a C program that finds all pairs of elements in an array whose sum equals 
a given target value


#include <stdio.h>

int main() {
    int arr[100], n, target, i, j, found = 0;

    printf("Enter size: ");
    scanf("%d", &n);

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) scanf("%d", &arr[i]);

    printf("Enter target sum: ");
    scanf("%d", &target);

    for (i = 0; i < n; i++) {
        for (j = i + 1; j < n; j++) {
            if (arr[i] + arr[j] == target) {
                printf("(%d, %d)\n", arr[i], arr[j]);
                found = 1;
            }
        }
    }

    if (!found) printf("No pairs found.\n");

    return 0;
}

26.  Write a C program that rearranges an array such that all negative numbers 
appear before all positive numbers while preserving the relative order of both 
groups.

#include <stdio.h>

int main() {
    int arr[100], neg[100], pos[100];
    int n, i, ni = 0, pi = 0;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    // Separate negatives and positives
    for (i = 0; i < n; i++) {
        if (arr[i] < 0)
            neg[ni++] = arr[i];
        else
            pos[pi++] = arr[i];
    }

    // Merge back into arr
    for (i = 0; i < ni; i++) arr[i] = neg[i];
    for (i = 0; i < pi; i++) arr[ni + i] = pos[i];

    printf("Rearranged array:\n");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}
